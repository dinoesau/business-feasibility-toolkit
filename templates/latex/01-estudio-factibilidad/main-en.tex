\subsection{Metrics}
\begin{equation}
  \mathrm{NPV} = \sum_{t=1}^{5}\frac{CF_t}{(1+r)^t} - I_0
  = \$\VANbase{}\ \text{\Moneda{}}
  \label{eq:van}
\end{equation}
\begin{equation}
  \mathrm{IRR} = r^* \mid \mathrm{NPV}(r^*) = 0 = \TIRbase{}\%
  \label{eq:tir}
\end{equation}
Payback period (simple): \Payback{} years.
Decision rule: GO if \(\mathrm{NPV} > 0\) and \(\mathrm{IRR} > r\).
